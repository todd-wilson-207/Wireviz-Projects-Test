\hypertarget{3-assembly-steps}{%
\subsection{3. Assembly Steps}\label{3-assembly-steps}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  Cut cables \textbf{W1} and \textbf{W2} to length and strip 1.5 mm on each conductor.
\item
  Crimp contacts onto each conductor and seat them in the housings.
\item
  Apply heat-shrink over the housing entries for strain relief.
\end{enumerate}
